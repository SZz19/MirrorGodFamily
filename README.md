#include <SPI.h>
#include <PortentaEthernet.h>
#include <Ethernet.h>

// ==========================================
// 1. 网络参数配置 (可根据你的电脑网段自行修改)
// ==========================================
IPAddress ip(192, 168, 1, 2);        // Opta 的静态 IP 地址
IPAddress dns(192, 168, 1, 1);       // 域名服务器 (直连时保持默认即可)
IPAddress gateway(192, 168, 1, 1);   // 默认网关 (直连时保持默认即可)
IPAddress subnet(255, 255, 255, 0);  // 子网掩码 (必须与电脑网卡完全一致)

void setup() {
  // ==========================================
  // 2. 硬件底层以太网初始化
  // ==========================================
  // 使用 Opta 专用的 __Ethernet 实例进行静态 IP 配置
  // 最后的参数 2000 代表超时时间为 2000 毫秒，防止在未插网线时 PLC 崩溃或无限卡死
  __Ethernet.begin(NULL, ip, dns, gateway, subnet, 2000);
}

void loop() {
  // ==========================================
  // 3. 保持为空
  // ==========================================
  // 硬件上的 PLC 运行时（Runtime）会自动在后台处理 Modbus TCP 等工业任务，
  // 此处不需要编写任何循环代码，留空即可。
}
